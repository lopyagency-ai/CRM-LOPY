create table leads (
  id uuid default gen_random_uuid() primary key,
  empresa text not null, nome_contato text, perfil text,
  cidade text, captacao text, gargalo text, reacao text,
  maturidade text, reuniao text, objecao text, resumo text,
  proximo_passo text, status text default 'Novo',
  responsavel text, created_at timestamptz default now()
);

create table reunioes (
  id uuid default gen_random_uuid() primary key,
  empresa text not null, data_reuniao date not null,
  hora text, descricao text, link text,
  responsavel text, notas text,
  created_at timestamptz default now()
);

alter table leads enable row level security;
create policy "Public access" on leads for all using (true);
